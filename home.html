const targetLangEl = document.getElementById('targetLang'); const swapBtn = document.getElementById('swapBtn'); let sse; let debounceTimer; swapBtn.addEventListener('click', () => { const a = sourceLangEl.value; sourceLangEl.value = targetLangEl.value; targetLangEl.value = a; triggerTranslate(); }); input.addEventListener('input', () => { // suggestions (non-streaming) clearTimeout(debounceTimer); debounceTimer = setTimeout(() => { const text = input.value.trim(); if (!text) { suggestionsEl.innerHTML = ''; cancelSSE(); output.textContent = ''; statusEl.textContent = 'idle'; return; } // fetch suggestions axios.post('/api/suggest', { text, sourceLang: sourceLangEl.value, targetLang: targetLangEl.value }) .then(res => { const arr = res.data.suggestions || []; suggestionsEl.innerHTML = arr.map(s => `
  • ${s.label}
    ${s.translated}
    • `).join(''); }).catch(err => { suggestionsEl.innerHTML = ''; }); // start streaming translation startSSE(text); }, 250); }); function cancelSSE() { if (sse) { try { sse.close(); } catch (e) {} sse = null; } } function startSSE(text) { cancelSSE(); output.textContent = ''; statusEl.textContent = 'conectando...'; // Use EventSource to connect to SSE const params = new URLSearchParams({ text, sourceLang: sourceLangEl.value, targetLang: targetLangEl.value }); const url = `/api/stream-translate?${params.toString()}`; // Note: EventSource uses GET; some servers/proxies may need special config for CORS/credentials. sse = new EventSource(url); sse.onopen = () => { statusEl.textContent = 'transcribiendo...'; }; sse.onmessage = (e) => { // Append chunk to output // The server escapes newlines inside chunk; unescape them const chunk = e.data.replace(/\\n/g, '\n'); output.textContent += chunk; }; sse.addEventListener('done', (e) => { statusEl.textContent = 'completado'; cancelSSE(); }); sse.onerror = (err) => { statusEl.textContent = 'error'; // console.error('SSE error', err); cancelSSE(); }; } function triggerTranslate() { const text = input.value.trim(); if (text) startSSE(text); }
